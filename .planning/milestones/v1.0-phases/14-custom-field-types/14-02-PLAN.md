Existing computeFormProgress (lib/form-builder/progress.ts):
```typescript
function isFilled(value: unknown): boolean {
  if (value === undefined || value === null) return false;
  if (typeof value === "string") return value.trim().length > 0;
  if (typeof value === "boolean") return value;
  if (Array.isArray(value)) return value.length > 0;
  if (typeof value === "number") return !Number.isNaN(value);
  return true;
}
export function computeFormProgress(schema, values): number {
  const requiredIds = Object.entries(schema.entities)
    .filter(([, e]) => e.attributes?.required === true).map(([id]) => id);
  if (requiredIds.length === 0) return 100;
  const filled = requiredIds.filter(id => isFilled(values[id])).length;
  return Math.round(filled / requiredIds.length * 100);
}
```
